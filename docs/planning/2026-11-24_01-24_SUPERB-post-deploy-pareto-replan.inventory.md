```python
import re, os, glob, collections
root='/home/lars/projects/webphone'
exts='go|templ|md|js|mjs|css|nix|sh|json|py|txt|yml|yaml|toml|input'
path_re=re.compile(r'(/home/lars/projects/[^\s`:()]+|(?:internal|cmd|scripts|docs|package|nix)/[^\s`:()]+)\.(?:'+exts+r'):([0-9]+)(?:[-–]([0-9]+))?')
bare_re=re.compile(r'(?<![\w/.-])([A-Za-z0-9_][A-Za-z0-9_.-]*\.(?:'+exts+r')):([0-9]+)(?:[-–]([0-9]+))?')
idx=collections.defaultdict(list)
for d,_,fs in os.walk(root):
    if '/.git' in d or '/node_modules' in d: continue
    for f in fs: idx[f].append(os.path.join(d,f))
def check(p,a,b):
    if not p.startswith('/'): p=os.path.join(root,p)
    if not os.path.isfile(p): return 'MISSING'
    L=sum(1 for _ in open(p,encoding='utf-8',errors='replace'))
    return 'OK' if b<=L else f'OUT>{L}'
stats=collections.Counter(); bad=[]
for f in sorted(glob.glob(root+'/docs/review/libs/*.md')):
    name=os.path.basename(f)
    for ln,line in enumerate(open(f,encoding='utf-8'),1):
        for m in path_re.finditer(line):
            p=m.group(0).rsplit(':',1)[0]; a,b=int(m.group(2)),int(m.group(3) or m.group(2))
            kind='abs' if p.startswith('/') else 'rel'
            r=check(p,a,b); stats[(kind,r)]+=1
            if r!='OK': bad.append((name,ln,m.group(0),r))
        for m in bare_re.finditer(line):
            s=m.start()
            if s>0 and line[s-1] in '/': continue
            p=m.group(1)
            cands=idx.get(p,[])
            if len(cands)!=1: stats[('bare','AMBIG' if cands else 'UNKNOWN')]+=1; continue
            r=check(cands[0],int(m.group(2)),int(m.group(3) or m.group(2))); stats[('bare',r)]+=1
            if r!='OK': bad.append((name,ln,m.group(0),r))
print(dict(sorted(stats.items())))
seen=set()
for b in bad:
    k=(b[0],b[2])
    if k in seen: continue
    seen.add(k); print('  BAD',b)
print('distinct bad:',len(seen))
```
